cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS" FORCE)
endif()
project(SpeedSyncSQL VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
find_package(Qt6 6.10.2 EXACT REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Sql Test)
qt_standard_project_setup()
add_library(foundation desktop/foundation.cpp desktop/credentials.cpp)
target_include_directories(foundation PUBLIC desktop)
target_link_libraries(foundation PUBLIC Qt6::Widgets Qt6::Sql)
if(APPLE)
  target_link_libraries(foundation PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(foundation PRIVATE Advapi32)
endif()
qt_add_executable(speed-sync-sql WIN32 MACOSX_BUNDLE desktop/main.cpp)
target_link_libraries(speed-sync-sql PRIVATE foundation Qt6::WebEngineWidgets Qt6::WebChannel)
set_target_properties(speed-sync-sql PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.speedsync.sql" MACOSX_BUNDLE_BUNDLE_NAME "Speed Sync SQL")
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/frontend/dist/index.html")
  message(FATAL_ERROR "Build frontend first: cd frontend && npm ci && npm run build")
endif()
file(GLOB_RECURSE UI_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/frontend/dist/*")
qt_add_resources(speed-sync-sql ui PREFIX "/ui" BASE "${CMAKE_SOURCE_DIR}/frontend/dist" FILES ${UI_FILES})
set(SPEED_SYNC_MYSQL_PLUGIN "" CACHE FILEPATH "QMYSQL plugin matching Qt and architecture")
if(SPEED_SYNC_MYSQL_PLUGIN)
  add_custom_command(TARGET speed-sync-sql POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:speed-sync-sql>/plugins/sqldrivers"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SPEED_SYNC_MYSQL_PLUGIN}" "$<TARGET_FILE_DIR:speed-sync-sql>/plugins/sqldrivers/")
endif()
enable_testing()
qt_add_executable(foundation-tests tests/foundation_test.cpp)
target_link_libraries(foundation-tests PRIVATE foundation Qt6::Test)
add_test(NAME foundation COMMAND foundation-tests)
set_tests_properties(foundation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(async-tests tests/async_test.cpp)
target_link_libraries(async-tests PRIVATE foundation Qt6::Test)
add_test(NAME async-processes COMMAND async-tests)
set_tests_properties(async-processes PROPERTIES TIMEOUT 20)
